function Card({ id, title, emoji, color, children, open, onToggle, depth = 0 }) {
  const isOpen = open.has(id);
  return (
    <div style={{
      marginLeft: depth * 14, marginBottom: 10,
      border: '1px solid ' + color + '55', borderRadius: 12,
      background: 'linear-gradient(135deg,rgba(255,255,255,0.03),rgba(255,255,255,0.01))',
      boxShadow: isOpen ? '0 0 20px ' + color + '33' : 'none', overflow: 'hidden',
    }}>
      <button onClick={() => onToggle(id)} style={{
        width: '100%', textAlign: 'left', padding: '12px 14px',
        background: isOpen ? color + '1a' : 'transparent', border: 'none',
        cursor: 'pointer', display: 'flex', alignItems: 'center', gap: 9, fontFamily: 'inherit',
      }}>
        <span style={{ fontSize: 18 }}>{emoji}</span>
        <span style={{ fontWeight: 700, fontSize: 14, color }}>{title}</span>
        <span style={{ marginLeft: 'auto', color, fontSize: 12 }}>{isOpen ? '▲' : '▼'}</span>
      </button>
      {isOpen && <div style={{ padding: '4px 16px 16px 16px' }}>{children}</div>}
    </div>
  );
}
